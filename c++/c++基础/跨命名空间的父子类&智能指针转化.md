## 父子类跨命名空间，还能继承和互相转化吗？

```
#include <iostream>
#include <memory>
using namespace std;

namespace parent{
	class base{
		public:
		base(){
			cout<<"parent base constructure"<<endl;
		}

		virtual void print(){
			cout<<"print parent"<<endl;
		}
	};
}

namespace child{
	class derived:public parent::base{
		public:
		derived(){
			cout<<"child derived constructure"<<endl;
		}

		void print() override{
			cout<<"print child"<<endl;
		}
	};
}


void test(std::shared_ptr<parent::base> ptr){
	ptr->print();
}

int main()
{
        // 跨命名空间子类指针赋值给父类普通指针
        parent::base *p_common = new child::derived();
	p_common->print();
	// 普通指针转为智能指针
	std::shared_ptr<parent::base> p_shared = std::shared_ptr<parent::base>(p_common);
	p_shared->print();
	// 智能指针转为普通指针
	p_common = p_shared.get();
	p_common->print();

	// 子类智能指针直接转为父类智能指针
	std::shared_ptr<child::derived> c = std::make_shared<child::derived>();
	test(c);

	//父类智能指针通过dynamic_pointer_cast转为子类智能指针
	//std::shared_ptr<parent::base> p_shared; 普通父类指针无法直接转为子类
	std::shared_ptr<parent::base> ptr = c;
        std::shared_ptr<child::derived> c_cvrt=std::dynamic_pointer_cast<child::derived>(c);
        c_cvrt->print();
        return 0;
}
```

输出

```
parent base constructure
child derived constructure
print child
parent base constructure
child derived constructure
print child
```
